Proyecto Base de Datos MySQL

Descripción del proyecto

Diseño de una base de datos para la gestión de productos, inventarios y ventas utilizando MySQL. Se implementaron relaciones entre tablas mediante claves primarias y foráneas, además de restricciones e índices para mejorar la organización de la información.

Estructura de la Base de Datos

Tabla tenants


CREATE TABLE tenants (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre_comercial VARCHAR(150) NOT NULL,
    nit VARCHAR(30) NULL,
    estado_suscripcion ENUM('activo','suspendido') NOT NULL DEFAULT 'activo',
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL
) ENGINE=InnoDB;

Tabla productos


CREATE TABLE productos (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    tenant_id BIGINT UNSIGNED NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    tipo ENUM('flor','cinta','envoltura','base') NOT NULL,
    precio_base DECIMAL(10,2) NOT NULL,
    stock INT NOT NULL DEFAULT 0,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    CONSTRAINT fk_productos_tenant 
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),

    INDEX idx_productos_tenant (tenant_id),

    UNIQUE KEY uq_producto_por_tenant 
    (tenant_id, nombre)

) ENGINE=InnoDB;

Tabla inventarios


CREATE TABLE inventarios (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    tenant_id BIGINT UNSIGNED NOT NULL,
    producto_id BIGINT UNSIGNED NOT NULL,
    cantidad INT NOT NULL DEFAULT 0,
    fecha_ingreso DATE NOT NULL,
    fecha_caducidad DATE NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    CONSTRAINT fk_inventarios_tenant 
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),

    CONSTRAINT fk_inventarios_producto 
    FOREIGN KEY (producto_id) REFERENCES productos(id)

) ENGINE=InnoDB;

Tabla ventas


CREATE TABLE ventas (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    tenant_id BIGINT UNSIGNED NOT NULL,
    usuario_id BIGINT UNSIGNED NOT NULL,
    total_bs DECIMAL(10,2) NOT NULL,
    metodo_pago ENUM('efectivo') NOT NULL DEFAULT 'efectivo',
    estado ENUM('pagada','anulada') NOT NULL DEFAULT 'pagada',
    fecha_venta DATETIME NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    CONSTRAINT fk_ventas_tenant 
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),

    CONSTRAINT fk_ventas_usuario 
    FOREIGN KEY (usuario_id) REFERENCES users(id)

) ENGINE=InnoDB;

Tabla detalle_ventas


CREATE TABLE detalle_ventas (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    tenant_id BIGINT UNSIGNED NOT NULL,
    venta_id BIGINT UNSIGNED NOT NULL,
    producto_id BIGINT UNSIGNED NOT NULL,
    cantidad INT NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    created_at TIMESTAMP NULL,
    updated_at TIMESTAMP NULL,

    CONSTRAINT fk_detalle_tenant 
    FOREIGN KEY (tenant_id) REFERENCES tenants(id),

    CONSTRAINT fk_detalle_venta 
    FOREIGN KEY (venta_id) REFERENCES ventas(id),

    CONSTRAINT fk_detalle_producto 
    FOREIGN KEY (producto_id) REFERENCES productos(id)

) ENGINE=InnoDB;
